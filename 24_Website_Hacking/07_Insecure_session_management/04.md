Here is how to implement the **refresh token rotation** pattern.

In this architecture, you split authentication into two tokens:

1. **Access Token:** Short-lived (e.g., 15 minutes) and stored securely **in memory** on the frontend.
2. **Refresh Token:** Longer-lived (e.g., 7 days) and stored safely on the backend in an **`HttpOnly` cookie** so JavaScript
   can never touch it. When the access token expires, the client uses the refresh token cookie to silently get a new access
   token.

---

### Backend: Node.js & Express Refresh Token Flow

This Express implementation handles login, issues both tokens, and provides a `/refresh` endpoint that rotates the refresh
token for added security.

```typescript
import express, { Request, Response } from "express";
import jwt from "jsonwebtoken";
import cookieParser from "cookie-parser";

const app = express();
app.use(express.json());
app.use(cookieParser());

const ACCESS_TOKEN_SECRET = process.env.ACCESS_TOKEN_SECRET || "access_secret";
const REFRESH_TOKEN_SECRET = process.env.REFRESH_TOKEN_SECRET || "refresh_secret";

// Mock database for stored refresh tokens (to enable revocation)
const refreshTokens: string[] = [];

app.post("/api/login", (req: Request, res: Response) => {
  const userId = "user_12345";

  // 1. Generate tokens
  const accessToken = jwt.sign({ userId }, ACCESS_TOKEN_SECRET, { expiresIn: "15m" });
  const refreshToken = jwt.sign({ userId }, REFRESH_TOKEN_SECRET, { expiresIn: "7d" });

  refreshTokens.push(refreshToken);

  // 2. Send refresh token in a secure HttpOnly cookie
  res.cookie("refreshToken", refreshToken, {
    httpOnly: true,
    secure: process.env.NODE_ENV === "production",
    sameSite: "strict",
    maxAge: 7 * 24 * 60 * 60 * 1000, // 7 days
  });

  // 3. Send access token in the response body (for in-memory storage)
  res.json({ accessToken });
});

app.post("/api/refresh", (req: Request, res: Response) => {
  const cookies = req.cookies;
  if (!cookies?.refreshToken) return res.sendStatus(401);

  const incomingRefreshToken = cookies.refreshToken;
  if (!refreshTokens.includes(incomingRefreshToken)) return res.sendStatus(403);

  jwt.verify(incomingRefreshToken, REFRESH_TOKEN_SECRET, (err: any, user: any) => {
    if (err) return res.sendStatus(403);

    // Rotate refresh token: remove old one, issue a new one
    const index = refreshTokens.indexOf(incomingRefreshToken);
    refreshTokens.splice(index, 1);

    const newAccessToken = jwt.sign({ userId: user.userId }, ACCESS_TOKEN_SECRET, { expiresIn: "15m" });
    const newRefreshToken = jwt.sign({ userId: user.userId }, REFRESH_TOKEN_SECRET, { expiresIn: "7d" });

    refreshTokens.push(newRefreshToken);

    res.cookie("refreshToken", newRefreshToken, {
      httpOnly: true,
      secure: process.env.NODE_ENV === "production",
      sameSite: "strict",
      maxAge: 7 * 24 * 60 * 60 * 1000,
    });

    res.json({ accessToken: newAccessToken });
  });
});
```

### Frontend: React Axios Interceptor for Silent Refresh

On the React side, you use an Axios response interceptor. If an API request fails with a `401 Unauthorized` error (meaning
the access token expired), the interceptor automatically calls `/api/refresh`, gets a new access token, updates the memory
state, and seamlessly retries the original request.

```typescript
import axios from "axios";

// Create a custom axios instance
export const api = axios.create({
  baseURL: "https://api.example.com",
  withCredentials: true, // Required to send the HttpOnly refresh token cookie automatically
});

let inMemoryAccessToken: string | null = null;

export const setAccessToken = (token: string | null) => {
  inMemoryAccessToken = token;
};

// 1. Request Interceptor: Attach access token to every outgoing request
api.interceptors.request.use(
  (config) => {
    if (inMemoryAccessToken) {
      config.headers.Authorization = `Bearer ${inMemoryAccessToken}`;
    }
    return config;
  },
  (error) => Promise.reject(error)
);

// 2. Response Interceptor: Catch 401 and refresh token automatically
api.interceptors.response.use(
  (response) => response,
  async (error) => {
    const originalRequest = error.config;

    // If request failed with 401 and we haven't tried retrying yet
    if (error.response?.status === 401 && !originalRequest._retry) {
      originalRequest._retry = true;

      try {
        // Call refresh endpoint to get a new access token using the HttpOnly cookie
        const response = await axios.post(
          "https://api.example.com/api/refresh",
          {},
          {
            withCredentials: true,
          }
        );

        const { accessToken } = response.data;
        setAccessToken(accessToken);

        // Retry the original request with the new access token
        originalRequest.headers.Authorization = `Bearer ${accessToken}`;
        return api(originalRequest);
      } catch (refreshError) {
        // If refresh fails (e.g. refresh token expired), log out the user
        setAccessToken(null);
        window.location.href = "/login";
        return Promise.reject(refreshError);
      }
    }

    return Promise.reject(error);
  }
);
```
