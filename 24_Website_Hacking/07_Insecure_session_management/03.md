Here are TypeScript snippets demonstrating how to implement token-based authentication using **JSON Web Tokens (JWT)** stored
securely in memory on the frontend (instead of vulnerable storage like `localStorage`), along with a secure cookie or
response setup on the Node.js backend.

---

### Backend: Node.js & Express JWT Issuance

In this example, the backend generates a short-lived JWT upon successful authentication and sends it securely back to the
client.

```typescript
import express, { Request, Response } from "express";
import jwt from "jsonwebtoken";

const app = express();
app.use(express.json());

const JWT_SECRET = process.env.JWT_SECRET || "super_secret_key";

app.post("/api/login", (req: Request, res: Response) => {
  const { username, password } = req.body;

  // Validate user credentials here...
  const userId = "user_12345";

  // Generate a short-lived JWT (e.g., 15 minutes)
  const token = jwt.sign({ userId }, JWT_SECRET, { expiresIn: "15m" });

  // Return the token in the response body so it can be stored in memory
  res.json({ token, message: "Authenticated successfully" });
});

// Protected route middleware
const authenticateToken = (req: Request, res: Response, next: Function) => {
  const authHeader = req.headers["authorization"];
  const token = authHeader && authHeader.split(" ")[1]; // Bearer <token>

  if (!token) return res.sendStatus(401);

  jwt.verify(token, JWT_SECRET, (err: any, user: any) => {
    if (err) return res.sendStatus(403);
    req.user = user;
    next();
  });
};

app.get("/api/protected", authenticateToken, (req: Request, res: Response) => {
  res.json({ message: "Access granted to sensitive data", user: (req as any).user });
});
```

### Frontend: React In-Memory Storage & Axios Interceptor

Storing tokens in `localStorage` or `sessionStorage` makes them vulnerable to XSS attacks because any malicious script can
read them. Storing the token in memory (via a React state or closure) protects it from script theft, though it will reset if
the page is refreshed.

```typescript
import React, { useState, createContext, useContext, useEffect } from 'react';
import axios from 'axios';

// 1. Create an Auth Context to hold the token in memory
interface AuthContextType {
  token: string | null;
  setToken: (token: string | null) => void;
}

const AuthContext = createContext<AuthContextType>({ token: null, setToken: () => {} });

export const AuthProvider: React.FC<{ children: React.ReactNode }> = ({ children }) => {
  const [token, setToken] = useState<string | null>(null);

  return (
    <AuthContext.Provider value={{ token, setToken }}>
      {children}
    </AuthContext.Provider>
  );
};

export const useAuth = () => useContext(AuthContext);

// 2. Configure Axios with an interceptor to attach the in-memory token
export const apiClient = axios.create({
  baseURL: 'https://api.example.com',
});

// We can attach a request interceptor dynamically or pass the token directly
export const createAuthenticatedClient = (token: string | null) => {
  const instance = axios.create({ baseURL: 'https://api.example.com' });

  instance.interceptors.request.use((config) => {
    if (token) {
      config.headers.Authorization = `Bearer ${token}`;
    }
    return config;
  });

  return instance;
};

```
